#include <WiFi.h>
#include <WebServer.h>
#include "DHT.h"
#include <ESPmDNS.h>
#include <Preferences.h>

#define DHTPIN 15
#define DHTTYPE DHT22

Preferences preferences;

// Credenciais de fallback (caso não tenha nada salvo)
String current_ssid = "NCA";
String current_password = "N21C30A19";

DHT dht(DHTPIN, DHTTYPE);
WebServer server(80);

void handleSensor() {
  float temperatura = dht.readTemperature();
  float umidade = dht.readHumidity();

  if (isnan(temperatura) || isnan(umidade)) {
    server.send(500, "application/json", "{\"error\":\"Erro ao ler DHT22\"}");
    Serial.println("[ERRO] Falha ao ler DHT22");
    return;
  }

  String json = "{";
  json += "\"temperature\":" + String(temperatura, 1) + ",";
  json += "\"humidity\":" + String(umidade, 1);
  json += "}";

  Serial.println("================================");
  Serial.println("LEITURA DO DHT22");
  Serial.println("================================");
  Serial.println("Temperatura: " + String(temperatura, 1) + " °C");
  Serial.println("Umidade: " + String(umidade, 1) + " %");
  Serial.println("JSON: " + json);

  server.send(200, "application/json", json);
}

  html += "<p>O sensor esta operando normalmente.</p>";

  server.send(200, "text/html", html);

void handleConfig() {
  String html = "<!DOCTYPE html><html lang='pt-BR'><head>";
  html += "<meta charset='UTF-8'>";
  html += "<meta name='viewport' content='width=device-width,initial-scale=1'>";

void handleSave() {
  if (server.hasArg("ssid") && server.hasArg("password")) {
    String new_ssid = server.arg("ssid");
    String new_pass = server.arg("password");

    // Salva na memória permanente
    preferences.begin("tcc_sensor", false);
    preferences.putString("ssid", new_ssid);
    preferences.putString("password", new_pass);
    preferences.end();

    String html = "<h2>Credenciais salvas!</h2><p>O sensor sera reiniciado agora para conectar na nova rede.</p>";
    server.send(200, "text/html", html);
    
    Serial.println("[REDE] Novas credenciais recebidas. Reiniciando ESP32 do sensor...");
    delay(2000); // Aguarda 2 segundos para o navegador carregar a resposta
    ESP.restart(); // Reinicia a placa para aplicar a nova rede
  } else {
    server.send(400, "text/plain", "Faltam dados.");
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println();
  Serial.println("================================");
  Serial.println("ESP32 + DHT22");
  Serial.println("================================");

  // Carrega configurações de Wi-Fi salvas
  preferences.begin("tcc_sensor", false);
  current_ssid = preferences.getString("ssid", "NCA");
  current_password = preferences.getString("password", "N21C30A19");
  preferences.end();

  dht.begin();

  WiFi.mode(WIFI_STA);
  WiFi.begin(current_ssid.c_str(), current_password.c_str());

  Serial.print("Conectando ao Wi-Fi: ");
  Serial.println(current_ssid);

  int tentativas = 0;
  while (WiFi.status() != WL_CONNECTED && tentativas < 40) {
    delay(500);
    Serial.print(".");
    tentativas++;
  }

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println();
    Serial.println("[ERRO] Nao foi possivel conectar ao Wi-Fi. Tente reiniciar a placa.");
    // Aqui podíamos criar um Access Point de emergência, mas para o TCC o Fallback é suficiente
  } else {
    Serial.println();
    Serial.println("[OK] Wi-Fi conectado!");
    Serial.print("IP da ESP32: ");
    Serial.println(WiFi.localIP());

    if (MDNS.begin("sensordht")) {
      Serial.println("MDNS iniciado: sensordht.local");
    }
  }

  // Rotas do Servidor Web
  server.on("/", HTTP_GET, handleRoot);
  server.on("/sensor", HTTP_GET, handleSensor);
  server.on("/config", HTTP_GET, handleConfig);
  server.on("/save", HTTP_POST, handleSave);

  server.begin();

  Serial.println("[OK] Servidor do DHT22 iniciado!");
  if(WiFi.status() == WL_CONNECTED) {
    Serial.print("Acesse: http://");
    Serial.println(WiFi.localIP());
    Serial.println("Ou acesse: http://sensordht.local");
  }
}

void loop() {
  server.handleClient();
}
