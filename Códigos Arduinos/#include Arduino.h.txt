// OBTER DADOS DO DHT22
// ============================================================

bool obterDadosDHT(float& temperatura, float& umidade) {

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("[DHT22] Wi-Fi desconectado.");
    return false;
  }

  WiFiClient client;
  HTTPClient http;

  String url = "http://" + String(dht_esp32_ip) + "/sensor";

  Serial.println();
  Serial.println("================================");
  Serial.println("CONSULTANDO DHT22");
  Serial.println("================================");

  Serial.print("[DHT22] URL: ");
  Serial.println(url);

  client.setTimeout(5000);

  http.setConnectTimeout(5000);
  http.setTimeout(5000);

  if (!http.begin(client, url)) {

    Serial.println(
      "[ERRO] Não foi possível iniciar conexão com DHT22."
    );

    return false;
  }

  Serial.println("[DHT22] Enviando GET /sensor...");

  int httpCode = http.GET();

  Serial.print("[DHT22] HTTP Code: ");
  Serial.println(httpCode);

  if (httpCode <= 0) {

    Serial.print("[ERRO] Falha na comunicação com DHT22: ");
    Serial.println(http.errorToString(httpCode));

    http.end();

    return false;
  }

  String resposta = http.getString();

  Serial.println("[DHT22] Resposta recebida:");
  Serial.println(resposta);

  if (httpCode != HTTP_CODE_OK) {

    Serial.print("[ERRO] DHT22 respondeu HTTP ");
    Serial.println(httpCode);

    http.end();

    return false;
  }

  // ==========================================================
  // LOCALIZA TEMPERATURA
  // ==========================================================

  int tempInicio = resposta.indexOf("\"temperature\":");

  // ==========================================================
  // LOCALIZA UMIDADE
  // ==========================================================

  int humInicio = resposta.indexOf("\"humidity\":");

  if (tempInicio == -1 || humInicio == -1) {

    Serial.println(
      "[ERRO] Campos temperature/humidity não encontrados."
    );

    http.end();

    return false;
  }

  tempInicio += strlen("\"temperature\":");
  humInicio += strlen("\"humidity\":");

  int tempFim = resposta.indexOf(",", tempInicio);
  int humFim = resposta.indexOf("}", humInicio);

  if (tempFim == -1 || humFim == -1) {

    Serial.println("[ERRO] Formato JSON inválido.");

    http.end();

    return false;
  }

  String tempString =
    resposta.substring(tempInicio, tempFim);

  String humString =
    resposta.substring(humInicio, humFim);

  temperatura = tempString.toFloat();
  umidade = humString.toFloat();

  http.end();

  Serial.println();
  Serial.println("[OK] DADOS DO DHT22 RECEBIDOS!");

  Serial.print("Temperatura: ");
  Serial.print(temperatura, 1);
  Serial.println(" °C");

  Serial.print("Umidade: ");
  Serial.print(umidade, 1);
  Serial.println(" %");

  Serial.println("================================");

  return true;
}


// ============================================================
// ENVIAR FOTO + DHT22 PARA DJANGO
// ============================================================

bool enviarParaDjango(
  camera_fb_t* fb,
  float temperatura,
  float umidade
) {

  if (WiFi.status() != WL_CONNECTED) {

    Serial.println("[ERRO] Wi-Fi desconectado.");

    return false;
  }

  WiFiClient client;
  HTTPClient http;

  Serial.println();
  Serial.println("================================");
  Serial.println("ENVIANDO CAPTURA PARA DJANGO");
  Serial.println("================================");

  if (!http.begin(client, apiUrl)) {

    Serial.println(
      "[ERRO] Não foi possível conectar ao Django."
    );

    return false;
  }

  String boundary = "----ESP32CAMBoundary";

  // ==========================================================
  // INÍCIO DO MULTIPART
  // ==========================================================

  String bodyStart;

  bodyStart += "--" + boundary + "\r\n";

  bodyStart +=
    "Content-Disposition: form-data; "
    "name=\"temperature\"\r\n\r\n";

  bodyStart += String(temperatura, 1);

  bodyStart += "\r\n";

  // ==========================================================
  // UMIDADE
  // ==========================================================

  bodyStart += "--" + boundary + "\r\n";

  bodyStart +=
    "Content-Disposition: form-data; "
    "name=\"humidity\"\r\n\r\n";

  bodyStart += String(umidade, 1);

  bodyStart += "\r\n";

  // ==========================================================
  // OBSERVAÇÃO
  // ==========================================================

  bodyStart += "--" + boundary + "\r\n";

  bodyStart +=
    "Content-Disposition: form-data; "
    "name=\"observation\"\r\n\r\n";

  bodyStart +=
    "Captura realizada pela ESP32-CAM";

  bodyStart += "\r\n";

  // ==========================================================
  // IMAGEM
  // ==========================================================

  bodyStart += "--" + boundary + "\r\n";

  bodyStart +=
    "Content-Disposition: form-data; "
    "name=\"image\"; filename=\"tomate.jpg\"\r\n";

  bodyStart +=
    "Content-Type: image/jpeg\r\n\r\n";

  // ==========================================================
  // FINAL DO MULTIPART
  // ==========================================================

  String bodyEnd;

  bodyEnd += "\r\n--";
  bodyEnd += boundary;
  bodyEnd += "--\r\n";

  size_t totalLength =
    bodyStart.length() +
    fb->len +
    bodyEnd.length();

  // ==========================================================
  // HEADERS
  // ==========================================================

  http.addHeader(
    "Content-Type",
    "multipart/form-data; boundary=" + boundary
  );

  http.addHeader(
    "Authorization",
    "Token " + String(authToken)
  );

  http.addHeader(
    "Content-Length",
    String(totalLength)
  );

  Serial.print("[Django] Imagem: ");
  Serial.print(fb->len);
  Serial.println(" bytes");

  Serial.print("[Django] Temperatura: ");
  Serial.print(temperatura, 1);
  Serial.println(" °C");

  Serial.print("[Django] Umidade: ");
  Serial.print(umidade, 1);
  Serial.println(" %");

  Serial.println("[Django] Enviando...");

  // ==========================================================
  // MONTA PAYLOAD
  // ==========================================================

  String payload;

  payload.reserve(totalLength);

  payload = bodyStart;

  for (size_t i = 0; i < fb->len; i++) {
    payload += (char)fb->buf[i];
  }

  payload += bodyEnd;

  // ==========================================================
  // POST
  // ==========================================================

  int httpCode = http.POST(
    (uint8_t*)payload.c_str(),
    payload.length()
  );

  Serial.print("[Django] HTTP Code: ");
  Serial.println(httpCode);

  if (httpCode > 0) {

    String resposta = http.getString();

    Serial.println();
    Serial.println("--- RESPOSTA DJANGO ---");
    Serial.println(resposta);
    Serial.println("-----------------------");

  } else {

    Serial.print(
      "[ERRO] Falha no envio para Django: "
    );

    Serial.println(
      http.errorToString(httpCode)
    );
  }

  http.end();

  return (
    httpCode >= 200 &&
    httpCode < 300
  );
}

// Esta função é chamada pelo Get Still do CameraWebServer.

// Ela recebe exatamente o framebuffer que o servidor original
// acabou de capturar.
//

bool processarCapturaTCC(camera_fb_t* fb) {

  if (fb == nullptr) {

    Serial.println(
      "[TCC] Framebuffer inválido."
    );

    return false;
  }

  Serial.println();
  Serial.println("================================");
  Serial.println("PROCESSANDO CAPTURA DO TCC");
  Serial.println("================================");

  Serial.print("[TCC] Imagem capturada: ");
  Serial.print(fb->len);
  Serial.println(" bytes");

  // ==========================================================
  // CONSULTA DHT22
  // ==========================================================

  float temperatura = 0.0;
  float umidade = 0.0;

  bool dhtOK =
    obterDadosDHT(
      temperatura,
      umidade
    );

  if (!dhtOK) {

    Serial.println(
      "[TCC] Não foi possível obter os dados do DHT22."
    );

    Serial.println(
      "[TCC] A imagem continuará sendo exibida "
      "pelo CameraWebServer."
    );

    return false;
  }

  // ==========================================================
  // ENVIA PARA DJANGO
  // ==========================================================

  bool djangoOK =
    enviarParaDjango(
      fb,
      temperatura,
      umidade
    );

  if (!djangoOK) {

    Serial.println(
      "[TCC] Falha ao enviar captura para Django."
    );

    return false;
  }

  Serial.println();
  Serial.println(
    "[OK] CAPTURA ENVIADA PARA DJANGO!"
  );

  Serial.println("================================");

  return true;
}

typedef struct {
  size_t size;   //number of values used for filtering
  size_t index;  //current value index
  size_t count;  //value count
  int sum;
  int *values;  //array to be filled with values
} ra_filter_t;

static ra_filter_t ra_filter;

static ra_filter_t *ra_filter_init(ra_filter_t *filter, size_t sample_size) {
  memset(filter, 0, sizeof(ra_filter_t));

  filter->values = (int *)malloc(sample_size * sizeof(int));
  if (!filter->values) {
    return NULL;
  }
  memset(filter->values, 0, sample_size * sizeof(int));

  filter->size = sample_size;
  return filter;
}

#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO
static int ra_filter_run(ra_filter_t *filter, int value) {
  if (!filter->values) {
    return value;
  }
  filter->sum -= filter->values[filter->index];
  filter->values[filter->index] = value;
  filter->sum += filter->values[filter->index];
  filter->index++;
  filter->index = filter->index % filter->size;
  if (filter->count < filter->size) {
    filter->count++;
  }
  return filter->sum / filter->count;
}
#endif

static esp_err_t bmp_handler(httpd_req_t *req) {
  camera_fb_t *fb = NULL;
  esp_err_t res = ESP_OK;
#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO
  int64_t fr_start = esp_timer_get_time();
#endif
  fb = esp_camera_fb_get();
  if (!fb) {
    log_e("Camera capture failed");
    httpd_resp_send_500(req);
    return ESP_FAIL;
  }

  httpd_resp_set_type(req, "image/x-windows-bmp");
  httpd_resp_set_hdr(req, "Content-Disposition", "inline; filename=capture.bmp");
  httpd_resp_set_hdr(req, "Access-Control-Allow-Origin", "*");

  char ts[32];
  // Cast to uint32_t is safe until year 2106.
  snprintf(ts, 32, "%" PRIu32 ".%06" PRIu32, (uint32_t)fb->timestamp.tv_sec, (uint32_t)fb->timestamp.tv_usec);
  httpd_resp_set_hdr(req, "X-Timestamp", (const char *)ts);

  uint8_t *buf = NULL;
  size_t buf_len = 0;
  bool converted = frame2bmp(fb, &buf, &buf_len);
  esp_camera_fb_return(fb);
  if (!converted) {
    log_e("BMP Conversion failed");
    httpd_resp_send_500(req);
    return ESP_FAIL;
  }
  res = httpd_resp_send(req, (const char *)buf, buf_len);
  free(buf);
#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO
  int64_t fr_end = esp_timer_get_time();
#endif
  log_i("BMP: %" PRId32 "ms, %" PRIu32 "B", (int32_t)((fr_end - fr_start) / 1000), (uint32_t)buf_len);
  return res;
}

static size_t jpg_encode_stream(void *arg, size_t index, const void *data, size_t len) {
  jpg_chunking_t *j = (jpg_chunking_t *)arg;
  if (!index) {
    j->len = 0;
  }
  if (httpd_resp_send_chunk(j->req, (const char *)data, len) != ESP_OK) {
    return 0;
  }
  j->len += len;
  return len;
}

static esp_err_t capture_handler(httpd_req_t *req) {

  camera_fb_t *fb = NULL;

  esp_err_t res = ESP_OK;

#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO

  int64_t fr_start =
    esp_timer_get_time();

#endif

  // ==========================================================
  // CAPTURA ORIGINAL DA CÂMERA
  // ==========================================================

#if defined(LED_GPIO_NUM)

  enable_led(true);

  vTaskDelay(
    150 / portTICK_PERIOD_MS
  );

  fb = esp_camera_fb_get();

  enable_led(false);

#else

  fb = esp_camera_fb_get();

#endif

  // ==========================================================
  // VERIFICA CAPTURA
  // ==========================================================

  if (!fb) {

    log_e(
      "Camera capture failed"
    );

    httpd_resp_send_500(req);

    return ESP_FAIL;
  }

  Serial.println();
  Serial.println(
    "================================"
  );

  Serial.println(
    "[CAMERA] NOVA CAPTURA"
  );

  Serial.println(
    "================================"
  );

  Serial.print(
    "[CAMERA] Tamanho: "
  );

  Serial.print(
    fb->len
  );

  Serial.println(
    " bytes"
  );

  // ==========================================================
  // PROCESSAMENTO DO TCC
  // ==========================================================
  //
  // Aqui:
  //
  // 1. Consulta o ESP32 + DHT22
  // 2. Obtém temperatura
  // 3. Obtém umidade
  // 4. Envia imagem + dados para Django
  //
  // A imagem NÃO é capturada novamente.
  // Usamos exatamente o framebuffer do CameraWebServer.
  //

  bool tccOK =
    processarCapturaTCC(fb);

  if (tccOK) {

    Serial.println(
      "[TCC] Captura processada com sucesso."
    );

  } else {

    Serial.println(
      "[TCC] Falha no processamento "
      "DHT22/Django."
    );

    Serial.println(
      "[TCC] A imagem ainda será "
      "exibida no navegador."
    );
  }

  // ==========================================================
  // RESPOSTA ORIGINAL DO CAMERA WEBSERVER
  // ==========================================================

  httpd_resp_set_type(
    req,
    "image/jpeg"
  );

  httpd_resp_set_hdr(
    req,
    "Content-Disposition",
    "inline; filename=capture.jpg"
  );

  httpd_resp_set_hdr(
    req,
    "Access-Control-Allow-Origin",
    "*"
  );

  // ==========================================================
  // TIMESTAMP
  // ==========================================================

  char ts[32];

  snprintf(
    ts,
    32,
    "%" PRIu32 ".%06" PRIu32,
    (uint32_t)fb->timestamp.tv_sec,
    (uint32_t)fb->timestamp.tv_usec
  );

  httpd_resp_set_hdr(
    req,
    "X-Timestamp",
    (const char *)ts
  );

  // ==========================================================
  // ENVIA IMAGEM
  // ==========================================================

#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO

  size_t fb_len = 0;

#endif

  if (
    fb->format ==
    PIXFORMAT_JPEG
  ) {

#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO

    fb_len = fb->len;

#endif

    res =
      httpd_resp_send(
        req,
        (const char *)fb->buf,
        fb->len
      );

  } else {

    jpg_chunking_t jchunk = {
      req,
      0
    };

    res =
      frame2jpg_cb(
        fb,
        80,
        jpg_encode_stream,
        &jchunk
      )
      ? ESP_OK
      : ESP_FAIL;

    httpd_resp_send_chunk(
      req,
      NULL,
      0
    );

#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO

    fb_len = jchunk.len;

#endif
  }

  // ==========================================================
  // LIBERA FRAMEBUFFER
  // ==========================================================

  esp_camera_fb_return(fb);

#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO

  int64_t fr_end =
    esp_timer_get_time();

#endif

  log_i(
    "JPG: %" PRIu32 "B %" PRId32 " ms",
    (uint32_t)fb_len,
    (int32_t)(
      (fr_end - fr_start) / 1000
    )
  );

  return res;
}

static esp_err_t stream_handler(httpd_req_t *req) {
  camera_fb_t *fb = NULL;
  struct timeval _timestamp;
  esp_err_t res = ESP_OK;
  size_t _jpg_buf_len = 0;
  uint8_t *_jpg_buf = NULL;
  char *part_buf[128];

  static int64_t last_frame = 0;
  if (!last_frame) {
    last_frame = esp_timer_get_time();
  }

  res = httpd_resp_set_type(req, _STREAM_CONTENT_TYPE);
  if (res != ESP_OK) {
    return res;
  }

  httpd_resp_set_hdr(req, "Access-Control-Allow-Origin", "*");
  httpd_resp_set_hdr(req, "X-Framerate", "60");

#if defined(LED_GPIO_NUM)
  isStreaming = true;
  enable_led(true);
#endif

  while (true) {
    fb = esp_camera_fb_get();
    if (!fb) {
      log_e("Camera capture failed");
      res = ESP_FAIL;
    } else {
      _timestamp.tv_sec = fb->timestamp.tv_sec;
      _timestamp.tv_usec = fb->timestamp.tv_usec;
      if (fb->format != PIXFORMAT_JPEG) {
        bool jpeg_converted = frame2jpg(fb, 80, &_jpg_buf, &_jpg_buf_len);
        esp_camera_fb_return(fb);
        fb = NULL;
        if (!jpeg_converted) {
          log_e("JPEG compression failed");
          res = ESP_FAIL;
        }
      } else {
        _jpg_buf_len = fb->len;
        _jpg_buf = fb->buf;
      }
    }
    if (res == ESP_OK) {
      res = httpd_resp_send_chunk(req, _STREAM_BOUNDARY, strlen(_STREAM_BOUNDARY));
    }
    if (res == ESP_OK) {
      size_t hlen = snprintf((char *)part_buf, 128, _STREAM_PART, _jpg_buf_len, _timestamp.tv_sec, _timestamp.tv_usec);
      res = httpd_resp_send_chunk(req, (const char *)part_buf, hlen);
    }
    if (res == ESP_OK) {
      res = httpd_resp_send_chunk(req, (const char *)_jpg_buf, _jpg_buf_len);
    }
    if (fb) {
      esp_camera_fb_return(fb);
      fb = NULL;
      _jpg_buf = NULL;
    } else if (_jpg_buf) {
      free(_jpg_buf);
      _jpg_buf = NULL;
    }
    if (res != ESP_OK) {
      log_e("Send frame failed");
      break;
    }
    int64_t fr_end = esp_timer_get_time();

    int64_t frame_time = fr_end - last_frame;
    last_frame = fr_end;

    frame_time /= 1000;
#if ARDUHAL_LOG_LEVEL >= ARDUHAL_LOG_LEVEL_INFO
    uint32_t avg_frame_time = ra_filter_run(&ra_filter, frame_time);
#endif
    log_i(
      "MJPG: %" PRIu32 "B %" PRId32 "ms (%.1ffps), AVG: %" PRIu32 "ms (%.1ffps)", (uint32_t)_jpg_buf_len, (int32_t)frame_time, 1000.0 / frame_time,
      avg_frame_time, 1000.0 / avg_frame_time
    );
  }

#if defined(LED_GPIO_NUM)
  isStreaming = false;
  enable_led(false);
#endif

  return res;
}

static esp_err_t parse_get(httpd_req_t *req, char **obuf) {
  char *buf = NULL;
  size_t buf_len = 0;

  buf_len = httpd_req_get_url_query_len(req) + 1;
  if (buf_len > 1) {
    buf = (char *)malloc(buf_len);
    if (!buf) {
      httpd_resp_send_500(req);
      return ESP_FAIL;
    }
    if (httpd_req_get_url_query_str(req, buf, buf_len) == ESP_OK) {
      *obuf = buf;
      return ESP_OK;
    }
    free(buf);
  }
  httpd_resp_send_404(req);
  return ESP_FAIL;
}

    res = s->set_agc_gain(s, val);

    res = s->set_aec2(s, val);

// Handler para alterar intervalo
static esp_err_t api_interval_handler(httpd_req_t *req) {
  char buf[100];
  int ret, remaining = req->content_len;
  if (remaining >= sizeof(buf)) return httpd_resp_send_500(req);
  ret = httpd_req_recv(req, buf, remaining);
  if (ret <= 0) return ESP_FAIL;
  buf[ret] = '\0';

  char val[16] = {0};
  if (httpd_query_key_value(buf, "interval", val, sizeof(val)) == ESP_OK) {
    aplicarNovoIntervalo(atoi(val));
    const char* resp = "Intervalo alterado com sucesso! Regresse ao painel.";
    httpd_resp_set_type(req, "text/plain; charset=utf-8");
    return httpd_resp_send(req, resp, strlen(resp));
  }
  return httpd_resp_send_404(req);
}

// Handler para alterar Wi-Fi
static esp_err_t api_wifi_handler(httpd_req_t *req) {
  char buf[200];
  int ret, remaining = req->content_len;
  if (remaining >= sizeof(buf)) return httpd_resp_send_500(req);
  ret = httpd_req_recv(req, buf, remaining);
  if (ret <= 0) return ESP_FAIL;
  buf[ret] = '\0';

  char ssid[64] = {0};
  char pass[64] = {0};
  if (httpd_query_key_value(buf, "ssid", ssid, sizeof(ssid)) == ESP_OK &&
      httpd_query_key_value(buf, "password", pass, sizeof(pass)) == ESP_OK) {
    
    agendarTesteWifi(String(ssid), String(pass));
    
    const char* resp = "Credenciais recebidas. A câmara vai tentar a ligação. Se falhar, reverterá para a anterior em 15 segundos.";
    httpd_resp_set_type(req, "text/plain; charset=utf-8");
    return httpd_resp_send(req, resp, strlen(resp));
  }
  return httpd_resp_send_404(req);
}
